```html
<script>
async function qike() {
    const response = await fetch('http://localhost:5000/api/qike', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
            ri_gan: '甲',
            ri_zhi: '子',
            yue_jiang: '丑',
            shi_chen: '子'
        })
    });
    
    const result = await response.json();
    
    if (result.success) {
        // 使用结果
        console.log('四课:', result.data.sike);
        console.log('三传:', result.data.sanchuan.三传);
        console.log('起法:', result.data.sanchuan.起法);
    }
}
</script>
```
