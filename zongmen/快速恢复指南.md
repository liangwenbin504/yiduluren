#!/usr/bin/env python
# -*- coding: utf-8 -*-
"""
快速匹配 - 简化提示词版本

import json
import re
import time
from datetime import datetime
import dashscope

API_KEY = '***REMOVED***'

print("加载数据...")
with open(r'd:\新建文件夹\仪度六壬择日\yiduluren\zongmen\pdf_extract\64_ke_full\batch_analysis\64_lessons_batch_analysis.json', 'r', encoding='utf-8') as f:

with open(r'd:\新建文件夹\仪度六壬择日\yiduluren\data\720_ke_li_matched.json', 'r', encoding='utf-8') as f:
    ke_li_data = json.load(f)

print(f"64 课：{len(ke_jing_data)} | 课例：{len(ke_li_data)}")
print("开始批量匹配...")

matched = {}
errors = []
start = time.time()

for i, (key, ke_li) in enumerate(ke_li_data.items()):
    # 简化提示词
    prompt = f"{ke_li['ri_gan_zhi']}{ke_li['yue']}{ke_li['shi']}{ke_li['yue_jiang']} 匹配课经 JSON:{{\"matched_ke_jing\":[\"课经\"],\"primary_ke_jing\":\"课经\"}}"
    
    try:
        resp = Generation.call(model='qwen-max', messages=[
            {'role': 'system', 'content': '大六壬专家'},
            {'role': 'user', 'content': prompt}
        ], timeout=60)
        
        if resp.status_code == 200:
            text = resp.output.get('text', '')
            m = re.search(r'\{.*\}', text, re.DOTALL)
            if m:
                result = json.loads(m.group())
                if result.get('primary_ke_jing') in ke_jing_data:
                    matched[key] = {**ke_li, 'qwen_matched': result}
                    print(f"[{i+1}/{len(ke_li_data)}] ✓", end=" ")
                else:
                    errors.append({'key': key, 'error': '课经无效'})
            else:
                errors.append({'key': key, 'error': '解析失败'})
        else:
            errors.append({'key': key, 'error': f'API:{resp.status_code}'})
    except Exception as e:
        errors.append({'key': key, 'error': str(e)})
    
    if (i+1) % 10 == 0:
        with open(r'd:\新建文件夹\仪度六壬择日\yiduluren\zongmen\match_progress_quick.json', 'w', encoding='utf-8') as f:
            json.dump({'completed': i+1, 'total': len(ke_li_data), 'matched': matched, 'errors': errors}, f, ensure_ascii=False, indent=2)
        print(f"[已保存 {i+1}/{len(ke_li_data)}]")
    
    time.sleep(0.5)

# 保存结果
with open(r'd:\新建文件夹\仪度六壬择日\yiduluren\zongmen\qwen_max_quick_results.json', 'w', encoding='utf-8') as f:
    json.dump({'timestamp': datetime.now().isoformat(), 'total': len(ke_li_data), 'matched': len(matched), 'errors': len(errors), 'results': matched}, f, ensure_ascii=False, indent=2)

### 方案 3：分批次处理

分成小批次处理，每批 100 个课例：

创建文件 `batch_100.py`:

```python
#!/usr/bin/env python
# -*- coding: utf-8 -*-
"""
分批处理 - 每批 100 个课例
"""

import json
import re
import time
from datetime import datetime
import dashscope
from dashscope import Generation

API_KEY = '***REMOVED***'
dashscope.api_key = API_KEY

# 加载数据
with open(r'd:\新建文件夹\仪度六壬择日\yiduluren\zongmen\pdf_extract\64_ke_full\batch_analysis\64_lessons_batch_analysis.json', 'r', encoding='utf-8') as f:
    ke_jing_data = {item['lesson_name']: item for item in json.load(f)['results']}

with open(r'd:\新建文件夹\仪度六壬择日\yiduluren\data\720_ke_li_matched.json', 'r', encoding='utf-8') as f:
    ke_li_data = json.load(f)

# 分批处理
batch_size = 100
total_batches = (len(ke_li_data) + batch_size - 1) // batch_size

print(f"总课例：{len(ke_li_data)} | 批次：{total_batches} | 每批：{batch_size}")

for batch_idx in range(total_batches):
    start_idx = batch_idx * batch_size
    end_idx = min(start_idx + batch_size, len(ke_li_data))
    
    print(f"\n处理批次 {batch_idx+1}/{total_batches} (课例 {start_idx+1}-{end_idx})")
    
    batch_results = {}
    for i, key in enumerate(list(ke_li_data.keys())[start_idx:end_idx]):
        ke_li = ke_li_data[key]
        prompt = f"{ke_li['ri_gan_zhi']}{ke_li['yue']}{ke_li['shi']}{ke_li['yue_jiang']} 匹配 JSON:{{\"matched_ke_jing\":[\"课经\"],\"primary_ke_jing\":\"课经\"}}"
        
        try:
            resp = Generation.call(model='qwen-max', messages=[
                {'role': 'system', 'content': '大六壬专家'},
                {'role': 'user', 'content': prompt}
            ], timeout=60)
            
            if resp.status_code == 200:
                text = resp.output.get('text', '')
                m = re.search(r'\{.*\}', text, re.DOTALL)
                if m:
                    result = json.loads(m.group())
                    if result.get('primary_ke_jing') in ke_jing_data:
                        batch_results[key] = {**ke_li, 'qwen_matched': result}
                        print(f"✓", end=" ")
        except:
            pass
        
        time.sleep(0.5)
    
    # 保存批次结果
    filename = f'd:\\新建文件夹\\仪度六壬择日\\yiduluren\\zongmen\\batch_result_{batch_idx+1:03d}.json'
    with open(filename, 'w', encoding='utf-8') as f:
        json.dump({'batch': batch_idx+1, 'results': batch_results}, f, ensure_ascii=False, indent=2)
    
    print(f"\n批次 {batch_idx+1} 完成：{len(batch_results)}/{end_idx-start_idx}")
    
    # 询问是否继续
    if batch_idx < total_batches - 1:
        cont = input("继续下一批？(y/n): ")
        if cont.lower() != 'y':
            break

print("\n所有批次完成！")
```

运行：
```bash
python batch_100.py
```

**优点**:
- 每批独立处理
- 可以随时暂停
- 易于调试
